<script>
    const todosItens = <%- todosItensJSON %>;
    const todosProcessos = <%- JSON.stringify(processos) %>; 
    const malhariaId = "<%= malharia.id %>";
    const maxInternos = parseInt("<%= malharia.num_internos || 0 %>");
    let maquinasDaMalharia = [];
    let listaProducao = [];
    let combinacoesDisponiveis = [];

    function toggleSemProducao() {
        const isSemProducao = document.getElementById('sem_producao').checked;
        const blocoProducao = document.getElementById('blocoProducao');
        const blocoJustificativa = document.getElementById('blocoJustificativa');
        const selectJustificativa = document.getElementById('justificativa_id');
        if (isSemProducao) {
            blocoProducao.style.display = 'none';
            blocoJustificativa.style.display = 'block';
            selectJustificativa.required = true;
            
            listaProducao = [];
            atualizarTabela(); 
            
            document.getElementById('qtd_internos').min = "0";
        } else {
            blocoProducao.style.display = 'block';
            blocoJustificativa.style.display = 'none';
            selectJustificativa.required = false;
            selectJustificativa.value = '';
            
            document.getElementById('qtd_internos').min = "1";
        }
    }

    function exibirAviso(msg) {
        document.getElementById('textoAviso').innerText = msg;
        new bootstrap.Modal(document.getElementById('modalAviso')).show();
    }

    function escapeHTML(str) {
        if (!str || str.trim() === '') return '-';
        return String(str).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
    }

    // MODIFICADO: Função de visualização adaptada para carregar imagens ou o iframe do PDF
    void function zoomImagem(url, titulo) {
        window.zoomImagem = function(url, titulo) {
            document.getElementById('modalZoomTitulo').innerText = titulo;
            const imgEl = document.getElementById('modalZoomElemento');
            const pdfContainer = document.getElementById('modalZoomPdfContainer');
            const pdfEl = document.getElementById('modalZoomPdfElemento');
            
            const isPdf = url && url.toLowerCase().split('?')[0].endsWith('.pdf');
            
            if (isPdf) {
                imgEl.classList.add('d-none');
                imgEl.src = '';
                pdfContainer.classList.remove('d-none');
                pdfEl.src = url;
            } else {
                pdfContainer.classList.add('d-none');
                pdfEl.src = '';
                imgEl.classList.remove('d-none');
                imgEl.src = url;
            }
            
            const modal = new bootstrap.Modal(document.getElementById('modalZoomImagem'));
            modal.show();
        }
    }();

    $(document).ready(function() {
        const inputInternos = document.getElementById('qtd_internos');
        inputInternos.placeholder = `Máximo permitido: ${maxInternos}`;
        inputInternos.max = maxInternos;
        carregarDadosMalharia();
    });

    async function carregarDadosMalharia() {
        try {
            const res = await fetch(`/registros/api/malharia-info/${malhariaId}`);
             const data = await res.json();
            maquinasDaMalharia = data.itens || [];
            recalcularMeta();
        } catch (e) {
            console.error("Erro ao procurar máquinas da malharia:", e);
        }
    }

    function validarInternos() {
        const input = document.getElementById('qtd_internos');
         const aviso = document.getElementById('avisoInternos');
        if(parseInt(input.value) > maxInternos) {
            aviso.innerHTML = `<i class="bi bi-x-circle"></i> O número não pode exceder os ${maxInternos} internos vinculados à oficina!`;
            input.value = maxInternos;
        } else {
            aviso.innerText = '';
        }
    }

    function abrirModalBusca(filtroInicial = '') {
        combinacoesDisponiveis = [];
        $('#checkAllModal').prop('checked', false);

        todosProcessos.forEach(p => {
            const statusConcluido = p.situacao === 'Concluído' || p.situacao === 'Finalizado';
            const semDemanda = p.totais && p.totais.faltante <= 0;
            
            if(!statusConcluido && !semDemanda && p.itens) {
                p.itens.forEach((pi, idx) => {
                     const dadosItem = todosItens.find(i => String(i.id) === String(pi.item_id));
                    if(dadosItem) {
                        const descricaoFinal = pi.descricao || dadosItem.descricao || '-';
                        const tamanhoFinal = pi.tamanho || dadosItem.tamanho || ''; 

                        let urlImagem = null;
                        if (p.imagens && p.imagens.length > 0) {
                            const imgProduto = p.imagens.find(img => String(img.processo_item_id) === String(pi.id));
                            const imgGeral = p.imagens.find(img => !img.processo_item_id);
                            urlImagem = imgProduto ? imgProduto.url_publica : (imgGeral ? imgGeral.url_publica : null);
                        }

                        combinacoesDisponiveis.push({
                            processo_id: p.id,
                            orgao: p.orgao || 'N/A',
                            item_id: dadosItem.id,
                            nome_produto: dadosItem.nome || 'Sem Nome',
                            descricao: descricaoFinal,
                            tamanho: tamanhoFinal,
                            url_imagem: urlImagem,
                            id_unico: `${p.id}_${dadosItem.id}_${pi.id || idx}`
                        });
                    }
                });
            }
        });
        
        renderizarModalProdutos();
        $('#modalBuscaProdutos').modal('show');
        if (filtroInicial) {
            setTimeout(() => { $('#inputBuscaProduto').val(filtroInicial); filtrarModalProdutos(); }, 300);
        } else {
            $('#inputBuscaProduto').val(''); filtrarModalProdutos();
        }
    }

    function renderizarModalProdutos() {
        const tbody = $('#tbodyModalProdutos');
        tbody.empty();
        if(combinacoesDisponiveis.length === 0) {
            tbody.append('<tr><td colspan="5" class="text-center text-muted py-3">Nenhum produto pendente encontrado nos processos.</td></tr>');
            return;
        }

        combinacoesDisponiveis.forEach(comb => {
            const jaAdicionado = listaProducao.some(lp => String(lp.processo_id) === String(comb.processo_id) && String(lp.item_id) === String(comb.item_id) && String(lp.tamanho || '').trim() === String(comb.tamanho || '').trim() && String(lp.descricao || '').trim() === String(comb.descricao || '').trim());
            if(!jaAdicionado) {
                // MODIFICADO: Identificação se o link da arte é um PDF para renderizar ícone correspondente
                const isPdf = comb.url_imagem && comb.url_imagem.toLowerCase().split('?')[0].endsWith('.pdf');
                const imgHtml = comb.url_imagem 
                    ? (isPdf 
                        ? `<div class="text-center border rounded bg-white p-1 shadow-sm d-flex align-items-center justify-content-center" style="width: 45px; height: 45px; cursor: pointer;" onclick="zoomImagem('${comb.url_imagem}', '${escapeHTML(comb.nome_produto)}')"><i class="bi bi-file-earmark-pdf-fill text-danger fs-4"></i></div>`
                        : `<img src="${comb.url_imagem}" class="img-thumbnail rounded" style="width: 45px; height: 45px; object-fit: cover; cursor: pointer;" onclick="zoomImagem('${comb.url_imagem}', '${escapeHTML(comb.nome_produto)}')">`)
                    : `<div class="text-muted d-flex align-items-center justify-content-center border rounded bg-light" style="width: 45px; height: 45px;"><i class="bi bi-camera small"></i></div>`;

                tbody.append(`
                    <tr>
                         <td class="text-center align-middle">
                            <input type="checkbox" class="form-check-input check-produto border-success" value="${comb.id_unico}">
                        </td>
                        <td class="text-center align-middle p-1">${imgHtml}</td>
                        <td class="busca-texto align-middle fw-bold text-dark">${escapeHTML(comb.orgao)}</td>
                        <td class="busca-texto align-middle text-primary fw-bold">${escapeHTML(comb.nome_produto)}</td>
                        <td class="busca-texto align-middle text-muted small">
                            ${comb.tamanho ? `<span class="badge bg-secondary me-1">Tam: ${escapeHTML(comb.tamanho)}</span>` : ''}
                            ${escapeHTML(comb.descricao)}
                        </td>
                    </tr>
                `);
            }
        });
    }

    function filtrarModalProdutos() {
        const buscaTexto = $('#inputBuscaProduto').val().toLowerCase().trim();
        const palavras = buscaTexto === '' ? [] : buscaTexto.split(/\s+/);
        $('#tbodyModalProdutos tr').each(function() {
            const textoLinha = $(this).find('.busca-texto').text().toLowerCase();
            const corresponde = palavras.every(palavra => textoLinha.includes(palavra));
            $(this).toggle(corresponde || palavras.length === 0);
        });
    }

    function toggleAllModal(source) { $('.check-produto:visible').prop('checked', source.checked); }

    function adicionarProdutosSelecionados() {
        const selecionados = $('.check-produto:checked');
        if(selecionados.length === 0) { exibirAviso("Selecione pelo menos um produto para o planejamento."); return; }
        
        selecionados.each(function() {
            const idUnicoSelecionado = $(this).val();
            const comb = combinacoesDisponiveis.find(c => c.id_unico === idUnicoSelecionado);
            if (comb) {
                listaProducao.push({
                     id_unico: comb.id_unico,
                    processo_id: comb.processo_id,
                    orgao: comb.orgao,
                    item_id: comb.item_id,
                    nomeProduto: comb.nome_produto,
                     descricao: comb.descricao,
                    tamanho: comb.tamanho,
                    url_imagem: comb.url_imagem,
                    qtd: 0 
                });
            }
        });
        atualizarTabela();
        $('#modalBuscaProdutos').modal('hide');
    }

    function removerProducao(index) { listaProducao.splice(index, 1); atualizarTabela(); }

    function atualizarTabela() {
        const tbody = document.querySelector('#tabelaProducao tbody');
        if(listaProducao.length === 0) {
            tbody.innerHTML = '<tr id="linhaVazia"><td colspan="5" class="text-center text-muted py-3">Nenhum produto selecionado para o planejamento.</td></tr>';
            recalcularMeta(); return;
        }
        tbody.innerHTML = '';
        listaProducao.forEach((prod, index) => {
            // MODIFICADO: Identificação se o link da arte é um PDF na montagem da tabela
            const isPdf = prod.url_imagem && prod.url_imagem.toLowerCase().split('?')[0].endsWith('.pdf');
            const imgHtml = prod.url_imagem 
                ? (isPdf 
                    ? `<div class="text-center border rounded bg-white p-1 shadow-sm d-flex align-items-center justify-content-center" style="width: 45px; height: 45px; cursor: pointer;" onclick="zoomImagem('${prod.url_imagem}', '${escapeHTML(prod.nomeProduto)}')" title="Abrir PDF da Arte"><i class="bi bi-file-earmark-pdf-fill text-danger fs-4"></i></div>`
                    : `<img src="${prod.url_imagem}" class="img-thumbnail rounded" style="width: 45px; height: 45px; object-fit: cover; cursor: pointer;" onclick="zoomImagem('${prod.url_imagem}', '${escapeHTML(prod.nomeProduto)}')">`)
                : `<div class="text-muted d-flex align-items-center justify-content-center border rounded bg-light" style="width: 45px; height: 45px;"><i class="bi bi-camera small"></i></div>`;

            tbody.innerHTML += `
                <tr>
                    <td class="text-center align-middle p-1">${imgHtml}</td>
                    <td class="align-middle fw-bold text-dark">
                        <input type="hidden" name="processos_ids[]" value="${prod.processo_id}">
                         ${escapeHTML(prod.orgao)}
                    </td>
                    <td class="align-middle text-primary fw-bold">
                        <input type="hidden" name="itens_ids[]" value="${prod.item_id}">
                         ${escapeHTML(prod.nomeProduto)}
                    </td>
                    <td class="align-middle text-muted small">
                        ${prod.tamanho ? `<span class="badge bg-secondary me-1">Tam: ${escapeHTML(prod.tamanho)}</span>` : ''}
                         ${escapeHTML(prod.descricao)}
                    </td>
                    <td class="text-center align-middle">
                        <button type="button" class="btn btn-sm btn-outline-danger border-0" onclick="removerProducao(${index})" title="Remover"><i class="bi bi-trash3-fill fs-6"></i></button>
                    </td>
                 </tr>
            `;
        });
        recalcularMeta();
    }

    function recalcularMeta() {
        const displayMeta = document.getElementById('displayMeta');
        const inputMeta = document.getElementById('meta_calculada');
        
        if(listaProducao.length === 0) {
            displayMeta.innerText = "0";
            inputMeta.value = 0; 
            return;
        }

        let explicacao = "CÁLCULO DE META AUTOMÁTICO:\n==============================================\n\n";
        if (maquinasDaMalharia.length === 0) {
            explicacao += "AVISO: A malharia selecionada NÃO possui máquinas no sistema.\nA meta calculada será 0.";
            displayMeta.innerText = "0"; 
            inputMeta.value = 0; 
            document.getElementById('textoExplicacaoMeta').innerText = explicacao; 
            return;
        }

        const produtosUnicosIds = [...new Set(listaProducao.map(p => String(p.item_id)))];
        let usoMaquinario = {};
        let alertas = [];

        produtosUnicosIds.forEach(itemId => {
            const dadosItem = todosItens.find(i => String(i.id) === itemId);
            if(dadosItem) {
                if(dadosItem.maquinas && dadosItem.maquinas.length > 0) {
                    dadosItem.maquinas.forEach(maq => {
                         const idMaq = String(maq.maquinario_id || (maq.maquina && maq.maquina.id));
                        if(idMaq && idMaq !== 'undefined') {
                            if(!usoMaquinario[idMaq]) usoMaquinario[idMaq] = [];
                            usoMaquinario[idMaq].push({ nome: dadosItem.nome, meta: parseFloat(dadosItem.meta_diaria) || 100 });
                        }
                    });
                } else {
                    alertas.push(`O produto '${dadosItem.nome}' NÃO tem máquinas exigidas no cadastro.`);
                 }
            }
        });
        let contagemMalharia = {};
        let nomesMaquinas = {};
        
        maquinasDaMalharia.forEach(maqFisica => {
            if(maqFisica.status !== 'indisponivel') {
                const idMaq = String(maqFisica.maquinario_id || (maqFisica.maquinario && maqFisica.maquinario.id));
                if (idMaq && idMaq !== 'undefined') {
                    contagemMalharia[idMaq] = (contagemMalharia[idMaq] || 0) + 1;
                     if(maqFisica.maquinario) nomesMaquinas[idMaq] = maqFisica.maquinario.nome;
                }
            }
        });
        let metaTotalGeral = 0;
        let maquinasCompativeisEncontradas = 0;

        for (const [maqId, qtdFisica] of Object.entries(contagemMalharia)) {
            if (usoMaquinario[maqId]) {
                maquinasCompativeisEncontradas++;
                const produtosQueUsam = usoMaquinario[maqId];
                const somaMetas = produtosQueUsam.reduce((acc, p) => acc + p.meta, 0);
                const mediaMetaPorMaquina = somaMetas / produtosQueUsam.length;
                const subtotal = mediaMetaPorMaquina * qtdFisica;
                metaTotalGeral += subtotal;

                const nomeDaMaquina = nomesMaquinas[maqId] || `Máquina ID ${maqId}`;
                explicacao += `>> ${nomeDaMaquina.toUpperCase()}\n`;
                explicacao += `   - Quantidade Física na Oficina: ${qtdFisica} un.\n`;
                explicacao += `   - Produtos que a utilizam: ${produtosQueUsam.map(p => p.nome).join(' e ')}\n`;
                explicacao += `   - Média de Production: ${mediaMetaPorMaquina.toFixed(2)} por máquina/dia\n`;
                explicacao += `   - Subtotal Gerado (${mediaMetaPorMaquina.toFixed(2)} x ${qtdFisica}): ${subtotal.toFixed(2)}\n\n`;
            }
        }

        if (maquinasCompativeisEncontradas === 0) alertas.push("A malharia NÃO possui nenhuma máquina compatível para fazer os produtos da tabela.");
        if (alertas.length > 0) {
            explicacao += "AVISOS DO SISTEMA:\n";
            alertas.forEach(a => explicacao += `- ${a}\n`);
            explicacao += "\n";
        }

        explicacao += `==============================================\nMETA FINAL DO DIA: ${Math.round(metaTotalGeral)}`;
        displayMeta.innerText = Math.round(metaTotalGeral);
        inputMeta.value = Math.round(metaTotalGeral);
        
        document.getElementById('explicacao_meta').value = explicacao;
        document.getElementById('textoExplicacaoMeta').innerText = explicacao;
    }

    function mostrarExplicacaoMeta() { new bootstrap.Modal(document.getElementById('modalMeta')).show(); }

    UX_Validations: function validarFormulario(e) {
        const isSemProducao = document.getElementById('sem_producao').checked;
        if(!isSemProducao && listaProducao.length === 0) {
